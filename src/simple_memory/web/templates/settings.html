📌 MCP 配置

将以下配置添加到你的 Claude Desktop 或其他 MCP 客户端

{
  "mcpServers": {
    "simple-memory": {
      "command": "python",
      "args": ["-m", "simple_memory.mcp_server"],
      "cwd": "{{ config.database.path | replace('./data/lancedb', '') or '.' }}"
    }
  }
}

或者使用 uv 运行:

{
  "mcpServers": {
    "simple-memory": {
      "command": "uv",
      "args": ["run", "simple-memory-mcp"]
    }
  }
}
